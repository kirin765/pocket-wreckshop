using System;
using System.Collections;
using System.Linq;
using System.Reflection;
using UnityEngine;
using UnityEngine.UI;
namespace PocketWreckshop.Editor {
public static class ManualHammerChecks {
static void SetSize(int w,int h){
 var asm=typeof(UnityEditor.Editor).Assembly;var sizesType=asm.GetType("UnityEditor.GameViewSizes");
 var singleton=typeof(UnityEditor.ScriptableSingleton<>).MakeGenericType(sizesType).GetProperty("instance").GetValue(null);
 var groupType=asm.GetType("UnityEditor.GameViewSizeGroupType");var group=sizesType.GetMethod("GetGroup").Invoke(singleton,new[]{Enum.Parse(groupType,"Android")});
 var sizeType=asm.GetType("UnityEditor.GameViewSize");var mode=asm.GetType("UnityEditor.GameViewSizeType");
 var size=Activator.CreateInstance(sizeType,new object[]{Enum.Parse(mode,"FixedResolution"),w,h,"Manual checks "+w});
 group.GetType().GetMethod("AddCustomSize").Invoke(group,new[]{size});
 int count=(int)group.GetType().GetMethod("GetTotalCount").Invoke(group,null);
 var view=UnityEditor.EditorWindow.GetWindow(asm.GetType("UnityEditor.GameView"));
 view.GetType().GetProperty("selectedSizeIndex",BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Instance).SetValue(view,count-1);view.Repaint();
}
static void Capture(string label,int w=450,int h=800){
 var cam=Camera.main;var old=cam.targetTexture;var rect=cam.rect;var aspect=cam.aspect;var active=RenderTexture.active;
 var rt=new RenderTexture(w,h,24);var tex=new Texture2D(w,h,TextureFormat.RGB24,false);
 try{cam.targetTexture=rt;cam.rect=rect;cam.aspect=aspect;Canvas.ForceUpdateCanvases();cam.Render();RenderTexture.active=rt;tex.ReadPixels(new Rect(0,0,w,h),0,0);tex.Apply();System.IO.File.WriteAllBytes("Documentation/Manual-Hammer-"+label+"-"+w+".png",tex.EncodeToPNG());}
 finally{cam.targetTexture=old;cam.rect=rect;cam.aspect=aspect;RenderTexture.active=active;UnityEngine.Object.Destroy(rt);UnityEngine.Object.Destroy(tex);}
}


public static IEnumerator Run(){
 var game=UnityEngine.Object.FindFirstObjectByType<WorkshopGame>();var s=game.Session.State;
 s.activePhone=10;s.phase=0;s.hp=100;s.lootPending=false;s.dropCash=0;s.lotteryWon=false;s.autoLevel=3;s.autoEnabled=true;
 foreach(var size in new[]{new Vector2Int(450,800),new Vector2Int(360,640),new Vector2Int(390,844)}){
  SetSize(size.x,size.y);yield return null;yield return null;game.ShowTab(WorkshopTab.Tools);
  if(game.ui.body.GetComponentsInChildren<Text>().Any(t=>t.text.ToLowerInvariant().Contains("auto")))throw new Exception("Auto Hammer controls remain");
  Capture("Tools",size.x,size.y);game.ShowTab(WorkshopTab.Smash);
  int hp=s.hp;yield return new WaitForSecondsRealtime(1.5f);
  if(s.hp!=hp||game.AutoCanRun||game.Session.CanAutoStrike)throw new Exception("Saved automation still strikes");
  game.Strike();if(s.hp!=hp)throw new Exception("Premature contact");yield return new WaitForSecondsRealtime(.35f);
  if(s.hp!=hp-1)throw new Exception("Manual power 1 failed");
 }
 System.IO.File.WriteAllText("Documentation/Manual-Hammer-Checks.json","{\"passed\":true,\"legacyAutoLevel\":3,\"legacyAutoEnabled\":true,\"idleSecondsPerViewport\":1.5,\"manualDamage\":1,\"portraitViewports\":[\"450x800\",\"360x640\",\"390x844\"]}");
 Debug.Log("MANUAL HAMMER PLAY CHECKS PASS");
}
}}
